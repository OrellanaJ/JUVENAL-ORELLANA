class EstrategiaMulta {

    calcularMulta(diasAtraso) {
        throw new Error("El método calcularMulta debe ser implementado");
    }
}


// ESTRATEGIA 1 — SOCIO INFANTIL
// No paga multa, solamente se bloquean nuevos préstamos

class MultaInfantil extends EstrategiaMulta {

    calcularMulta(diasAtraso) {
        return 0;
    }
}


// ============================================================
// ESTRATEGIA 2 — SOCIO ADULTO
// 2 Bs por cada día de atraso.
// ============================================================

class MultaAdulto extends EstrategiaMulta {

    calcularMulta(diasAtraso) {
        return diasAtraso * 2;
    }
}


// ============================================================
// ESTRATEGIA 3 — SOCIO TERCERA EDAD
// 1 Bs por día con un máximo de 20 Bs.
// ============================================================

class MultaTerceraEdad extends EstrategiaMulta {

    calcularMulta(diasAtraso) {

        let multa = diasAtraso * 1;

        if (multa > 20) {
            multa = 20;
        }

        return multa;
    }
}


// ============================================================
// CONTEXTO
// Utiliza una estrategia sin conocer sus detalles internos.
// ============================================================

class CalculadorMulta {

    constructor(estrategia) {
        this.estrategia = estrategia;
    }

    calcular(diasAtraso) {
        return this.estrategia.calcularMulta(diasAtraso);
    }
}
